function doGet(e) {
  const parametros = e && e.parameter ? e.parameter : {};
  const origem = String(parametros.origem || "").trim().toLowerCase();

  Logger.log(JSON.stringify({ origemRecebida: origem, parametros: parametros }));

  if (origem === "admin") {
    return carregarAplicacaoAdministrativa();
  }

  return carregarAplicacaoPublica();
}

function prepararTemplateAplicacao(template) {
  template.urlSistema = CONFIG.WEB_APP.URL_PUBLICA;
  template.urlAdmin = obterUrlAdmin();
  template.sistemaNome = CONFIG.SISTEMA.NOME;
  template.sistemaVersao = CONFIG.SISTEMA.VERSAO;
  return template;
}

function finalizarTemplate(template, titulo) {
  const output = template.evaluate().setTitle(titulo || CONFIG.SISTEMA.NOME);
  if (CONFIG.WEB_APP && CONFIG.WEB_APP.PERMITIR_IFRAME === true) {
    output.setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  }
  return output;
}

function finalizarHtml(html, titulo) {
  const output = HtmlService.createHtmlOutput(String(html || ""))
    .setTitle(titulo || CONFIG.SISTEMA.NOME);
  if (CONFIG.WEB_APP && CONFIG.WEB_APP.PERMITIR_IFRAME === true) {
    output.setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  }
  return output;
}

function carregarAplicacaoPublica() {
  const template = prepararTemplateAplicacao(
    HtmlService.createTemplateFromFile("index")
  );
  template.mostrarModalAcesso = false;
  return finalizarTemplate(template, CONFIG.SISTEMA.NOME);
}

function carregarAplicacaoAdministrativa() {
  const email = AuthService.obterEmailAtivo();

  if (!email) {
    return renderizarPaginaErro("ErroEmailInvalido", "E-mail inválido");
  }

  if (!emailInstitucional(email)) {
    return renderizarPaginaErro("ErroAcessoRestrito", "Acesso restrito");
  }

  const resultado = obterOuCadastrarUsuario(email);

  if (!resultado || resultado.erro) {
    const mensagem = resultado && resultado.mensagem ? resultado.mensagem : "Erro ao consultar a aba USUARIOS.";
    return finalizarHtml("<h2>" + escaparHtmlServidor(mensagem) + "</h2>", "Erro");
  }

  AuthService.salvarSessao(email, resultado.perfil || CONFIG.PERFIS.USUARIO_CONSULTA);

  const template = prepararTemplateAplicacao(
    HtmlService.createTemplateFromFile("index")
  );

  // Só mostra o modal de solicitação se o perfil for USUARIO_CONSULTA
  template.mostrarModalAcesso = (resultado.perfil === CONFIG.PERFIS.USUARIO_CONSULTA);

  return finalizarTemplate(template, CONFIG.SISTEMA.NOME);
}

function escaparHtmlServidor(valor) {
  return String(valor === null || valor === undefined ? "" : valor)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#039;");
}

function obterUrlAdmin() {
  const base = String(CONFIG.WEB_APP && CONFIG.WEB_APP.URL_ADMIN ? CONFIG.WEB_APP.URL_ADMIN : "").trim();
  if (!base) throw new Error("URL_ADMIN não foi configurada no CONFIG.");
  if (base.includes("ID_ADMINISTRATIVO") || base.includes("ID_PUBLICO")) {
    throw new Error("Substitua os IDs fictícios pelas URLs reais das implantações.");
  }
  if (/[?&]origem=/i.test(base)) {
    return base.replace(/([?&])origem=[^&]*/i, "$1origem=admin");
  }
  const separador = base.includes("?") ? "&" : "?";
  return base + separador + "origem=admin";
}

function obterUrlAlternarConta() {
  return "https://accounts.google.com/logout?continue=" + encodeURIComponent(obterUrlAdmin());
}

function renderizarPaginaErro(nomeArquivo, titulo) {
  const template = HtmlService.createTemplateFromFile(nomeArquivo);
  template.urlAlternarConta = obterUrlAlternarConta();
  template.sistemaNome = CONFIG.SISTEMA.NOME;
  template.sistemaVersao = CONFIG.SISTEMA.VERSAO;
  return finalizarTemplate(template, titulo);
}

function include(nomeArquivo) {
  const nome = textoSeguro(nomeArquivo);
  if (!nome) throw new Error("Nome do arquivo não informado.");

  const template = HtmlService.createTemplateFromFile(nome);
  template.sistemaNome = CONFIG.SISTEMA.NOME;
  template.sistemaVersao = CONFIG.SISTEMA.VERSAO;
  template.urlSistema = CONFIG.WEB_APP.URL_PUBLICA;
  template.urlAdmin = obterUrlAdmin();
  template.mostrarModalAcesso = false;
  return template.evaluate().getContent();
}

/**
 * Localiza o usuário na aba USUARIOS.
 * Se for um e‑mail institucional ainda não cadastrado,
 * cadastra automaticamente com perfil USUARIO_CONSULTA e ATIVO = SIM.
 */
function obterOuCadastrarUsuario(email) {
  const perfilConsulta = CONFIG.PERFIS.USUARIO_CONSULTA;
  const emailNormalizado = normalizarEmail(email);

  if (!emailNormalizado) {
    return { erro: "EMAIL_INVALIDO", perfil: perfilConsulta };
  }

  try {
    const sheet = DB.usuarios();
    const mapa = DB.map(sheet);
    const idxEmail = mapa.EMAIL;
    if (!idxEmail) {
      return { erro: "PLANILHA", mensagem: "Aba USUARIOS não possui a coluna EMAIL.", perfil: perfilConsulta };
    }

    const dados = DB.read(sheet);
    const perfisValidos = [CONFIG.PERFIS.GESTOR_SISTEMA, CONFIG.PERFIS.GESTOR_CONTEUDO, CONFIG.PERFIS.USUARIO_CONSULTA];

    for (const linha of dados) {
      const emailLinha = normalizarEmail(linha[idxEmail - 1]);
      if (emailLinha !== emailNormalizado) continue;

      const idxNome = mapa.NOME;
      const idxPerfil = mapa.PERFIL;
      const idxAtivo = mapa.ATIVO;
      const nome = textoSeguro(idxNome ? linha[idxNome - 1] : "");
      const perfilInformado = String(idxPerfil ? linha[idxPerfil - 1] : perfilConsulta).trim().toUpperCase();
      const perfilSeguro = perfisValidos.includes(perfilInformado) ? perfilInformado : perfilConsulta;
      const ativo = paraBoolean(idxAtivo ? linha[idxAtivo - 1] : false);

      return {
        email: emailNormalizado,
        nome: nome || emailNormalizado.split("@")[0],
        perfil: ativo ? perfilSeguro : perfilConsulta,
        ativo: ativo
      };
    }

    // Usuário NÃO encontrado → cadastra automaticamente
    const lock = LockService.getScriptLock();
    lock.waitLock(10000);
    try {
      // Releitura para evitar duplicação em caso de concorrência
      const dadosAtuais = DB.read(sheet);
      for (const linha of dadosAtuais) {
        if (normalizarEmail(linha[idxEmail - 1]) === emailNormalizado) {
          return {
            email: emailNormalizado,
            nome: emailNormalizado.split("@")[0],
            perfil: perfilConsulta,
            ativo: true
          };
        }
      }

      const idxNome = mapa.NOME || 2;
      const idxPerfil = mapa.PERFIL || 3;
      const idxAtivo = mapa.ATIVO || 4;
      const cabecalhos = DB.headers(sheet);
      const novaLinha = [];
      for (let c = 0; c < cabecalhos.length; c++) novaLinha.push("");
      novaLinha[idxEmail - 1] = emailNormalizado;
      if (idxNome) novaLinha[idxNome - 1] = emailNormalizado.split("@")[0];
      if (idxPerfil) novaLinha[idxPerfil - 1] = perfilConsulta;
      if (idxAtivo) novaLinha[idxAtivo - 1] = "SIM";

      sheet.appendRow(novaLinha);
      SpreadsheetApp.flush();
    } finally {
      lock.releaseLock();
    }

    return {
      email: emailNormalizado,
      nome: emailNormalizado.split("@")[0],
      perfil: perfilConsulta,
      ativo: true
    };
  } catch (erro) {
    return { erro: "PLANILHA", mensagem: erro.message || String(erro), perfil: perfilConsulta };
  }
}
